118{

337 GLtexture ResourceManager::CreateOpenGLTexture(const char* path)
338 {
339 GLtexture texture;
340 i32 w, h, channels;
341 unsigned char* pixels = stbi_load(path, &w, &h, &channels, 4);
342 if (!pixels) {
343 const std::string log = stbi_failure_reason();
344 throw std::runtime_error("Failed to load image: " + log);
345 }
346
347 texture.w = w; texture.h = h;
348
349 glGenTextures(1, &texture.id);
350 glBindTexture(GL_TEXTURE_2D, texture.id);
351 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
352 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
353 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
354 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
355
356 glPixelStorei(GL_UNPACK_ALIGNMENT, 1);
357 glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA8, w, h, 0, GL_RGBA, GL_UNSIGNED_BYTE, pixels);
358
359 stbi_image_free(pixels);
360 return texture;
361 }

372 void ResourceManager::processNode(const uint32_t meshId, const aiScene* scene, aiNode* node)
373 {
374 auto& newMesh = m_Meshes.at(meshId);
375 for (unsigned int i = 0; i < node->mNumMeshes; i++) {
376 aiMesh* mesh = scene->mMeshes[node->mMeshes[i]];
377 aiColor3D color(1.0f, 1.0f, 1.0f);
378 aiMaterial* material = scene->mMaterials[mesh->mMaterialIndex];
379
380 std::vector<unsigned int> indexBuffer;
381 std::vector<Vertex> vertices;
382 vertices.reserve(mesh->mNumVertices);
383
384 bool hasNormals = mesh->HasNormals();
385 if (!hasNormals) {
386 A_WARN("Mesh vertex has no normals");
387 }
388 material->Get(AI_MATKEY_COLOR_DIFFUSE, color);
389 for (unsigned int j = 0; j < mesh->mNumVertices; j++) {
390 Vertex vertex{};
391 aiVector3D worldPos = mesh->mVertices[j];
392 vertex.pos = glm::vec3(worldPos.x, worldPos.y, worldPos.z);
393
394 if (hasNormals) {
395 vertex.normal = glm::vec3(mesh->mNormals[j].x, mesh->mNormals[j].y, mesh->mNormals[j].z);
396 }
397
398 if (mesh->HasTextureCoords(0)) {
399 vertex.uv = glm::vec2(mesh->mTextureCoords[0][j].x, mesh->mTextureCoords[0][j].y);
400 }
401 vertex.color = glm::vec3(color.r, color.g, color.b);
402 vertices.push_back(vertex);
403 }
404
405 auto& vb = newMesh.vertexbuffer;
406 auto& ib = newMesh.indexBuffer;
407
408 const uint32_t baseVertex = static_cast<uint32_t>(vb.size());
409
410 vb.insert(vb.end(), vertices.begin(), vertices.end());
411
412 for (unsigned int k = 0; k < mesh->mNumFaces; k++) {
413 const aiFace& f = mesh->mFaces[k];
414 for (unsigned int t = 0; t < f.mNumIndices; t++) {
415 ib.push_back(baseVertex + f.mIndices[t]);
416 }
417 }
418 }
419 for (int i = 0; i < node->mNumChildren; i++) {
420 processNode(meshId, scene, node->mChildren[i]);
421 }
422 }

424 void ResourceManager::loadScene(const std::string& path) {
425 Assimp::Importer importer;
426 const aiScene* scene = importer.ReadFile(
427 path.c_str(),
428 aiProcess_Triangulate |
429 aiProcess_ConvertToLeftHanded |
430 aiProcess_FlipUVs |
431 aiProcess_GenNormals
432 );
433
434 if (!scene) {
435 throw std::runtime_error(importer.GetErrorString());
436 }
437
438 std::string meshName = Filesystem::getFileName(path);
439 if (m_MeshIDbyName.contains(meshName)) {
440 A_DEBUG("Mesh '{}' already exists, skipping load", meshName);
441 return;
442 }
443
444 const uint32_t id = m_NextMeshID++;
445 const deviceType dt = Filesystem::findDeviceTypeByPath(path);
446
447 m_Meshes[id] = Andromeda::Mesh{};
448
449 m_ModelRecords[id] = ModelRecord{id,meshName,dt};
450 m_ModelIndexList.push_back(id);
451
452 m_MeshIDbyName.emplace(meshName, id);
453
454 aiNode* rootNode = scene->mRootNode;
455 processNode(id, scene, rootNode);
456 }
