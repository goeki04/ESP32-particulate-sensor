43 void throwShaderLog(GLuint objectID, const char* stage, GLenum statusType) {
44 GLint success = 0;
45 if (statusType == GL_COMPILE_STATUS) {

187 std::vector<ReflectedUniform> OpenGLContext::getProgramUniforms(ShaderProgramHandle handle)
188 {
189 std::vector<ReflectedUniform> result;
190
191 GLint numUniforms = 0;
192 glGetProgramiv(handle.apiID, GL_ACTIVE_UNIFORMS, &numUniforms);
193
194 GLint maxNameLength = 0;
195 glGetProgramiv(handle.apiID, GL_ACTIVE_UNIFORM_MAX_LENGTH, &maxNameLength);
196 std::vector<GLchar> nameBuffer(maxNameLength);
197
198 for (GLint i = 0; i < numUniforms; ++i) {
199 GLsizei length = 0;
200 GLint size = 0;
201 GLenum type = 0;
202
203 glGetActiveUniform(handle.apiID, i, maxNameLength, &length, &size, &type, nameBuffer.data());
204 std::string name(nameBuffer.data(), length);
205
206 GLint location = glGetUniformLocation(handle.apiID, name.c_str());
207 if (location == -1) continue;
208
209 ShaderDataType sType;
210 switch (type) {
211 case GL_FLOAT: sType = ShaderDataType::Float; break;
212 case GL_INT: sType = ShaderDataType::Int; break;
213 case GL_FLOAT_VEC2: sType = ShaderDataType::Vec2; break;
214 case GL_FLOAT_VEC3: sType = ShaderDataType::Vec3; break;
215 case GL_FLOAT_VEC4: sType = ShaderDataType::Vec4; break;
216 case GL_FLOAT_MAT4: sType = ShaderDataType::Mat4; break;
217 case GL_SAMPLER_2D: sType = ShaderDataType::Int; break;
218 default: continue;
219 }
220
221 result.push_back({ name, static_cast<u32>(location), sType });
222 }

331 void OpenGLContext::blitFramebuffer(std::shared_ptr<IFramebuffer> source, std::shared_ptr<IFramebuffer> target, bool copyDepth)
332 {
333 assert(source && "RHI Error: Source Framebuffer for blit is null!");
334
335 auto glSource = std::static_pointer_cast<GLFramebuffer>(source);
336 const auto& srcSpecs = source->getSpecification();
337
338 u32 targetID = 0;
339 u32 targetWidth = srcSpecs.width;
340 u32 targetHeight = srcSpecs.height;
341
342 if (target) {
343 auto glTarget = std::static_pointer_cast<GLFramebuffer>(target);
344 targetID = glTarget->getFramebufferID();
345 const auto& dstSpecs = target->getSpecification();
346 targetWidth = dstSpecs.width;
347 targetHeight = dstSpecs.height;
348 }
349
350 glBindFramebuffer(GL_READ_FRAMEBUFFER, glSource->getFramebufferID());
351 glBindFramebuffer(GL_DRAW_FRAMEBUFFER, targetID);
352
353 GLbitfield mask = GL_COLOR_BUFFER_BIT;
354 GLenum filter = GL_NEAREST;
355
356 if (copyDepth) {
357 mask |= GL_DEPTH_BUFFER_BIT;
358 filter = GL_NEAREST;
359 }
360
361 glBlitFramebuffer(
362 0, 0, srcSpecs.width, srcSpecs.height,
363 0, 0, targetWidth, targetHeight,
364 mask, filter
365 );
366 glBindFramebuffer(GL_FRAMEBUFFER, targetID);
367 }
