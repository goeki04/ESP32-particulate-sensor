103 bool Editor::RayIntersectAABB(const amath::CameraData& cam, const ECS::Component::AABB& aabb, const glm::mat4& modelMatrix)
104 {
105 const auto& [origin, direction] = cam.cursorToWorldRay;
106
107 const mat4 invModel = glm::inverse(modelMatrix);
108
109 auto o = glm::vec3(invModel * vec4(origin, 1.0f));
110 vec3 d = glm::normalize(vec3(invModel * vec4(direction, 0.0f)));
111
112 float tMin = 0.001f;
113 float tMax = 1e30f;
114
115 for (int i = 0; i < 3; ++i)
116 {
117 float oi = o[i];
118 float di = d[i];
119
120 if (std::fabs(di) < 1e-8f) {
121 if (oi < aabb.min[i] || oi > aabb.max[i]) return false;
122 continue;
123 }
124
125 float invD = 1.0f / di;
126 float t0 = (aabb.min[i] - oi) * invD;
127 float t1 = (aabb.max[i] - oi) * invD;
128 if (t0 > t1) std::swap(t0, t1);
129
130 tMin = std::max(tMin, t0);
131 tMax = std::min(tMax, t1);
132 if (tMin > tMax) return false;
133 }
134 return true;
135 }

163 amath::Ray Editor::cursorToWorldRay(const amath::CameraData& cam) {
164 const float x = (2.0f * cam.imGuiMouseX) / cam.framebufferSize.x - 1.0f;
165 const float y = 1.0f - (2.0f * cam.imGuiMouseY) / cam.framebufferSize.y;
166 const vec4 rayClip(x, y, -1.0f, 1.0f);
167
168 vec4 rayView = amath::inverse(cam.projection) * rayClip;
169 rayView = vec4(rayView.x, rayView.y, -1.0f, 0.0f);
170 const vec4 rayDir4 = amath::inverse(cam.viewMatrix) * rayView;
171 amath::Ray ray{ vec3(0.0f),vec3(0.0f) };
172
173 ray.direction = amath::normalize(vec3(rayDir4));
174 ray.origin = cam.cameraPos;
175 return ray;
176 }

184 void Editor::cameraMovement(amath::CameraData& cam)
185 {
186 if (cam.canRotate == false) {
187 return;
188 }
189
190 SDL_Window* currentWindow = SDL_GL_GetCurrentWindow();
191 const Uint32 mouseState = SDL_GetMouseState(nullptr, nullptr);
192 const bool rightMouseDown = mouseState & SDL_BUTTON_MASK(SDL_BUTTON_RIGHT);
193
194 static bool wasRightMouseDown = false;
195
196 if (rightMouseDown && !wasRightMouseDown) {
197 SDL_GetMouseState(&cam.lastMouseX, &cam.lastMouseY);
198 SDL_SetWindowRelativeMouseMode(currentWindow, true);
199 }
200 else if (!rightMouseDown && wasRightMouseDown) {
201 SDL_WarpMouseInWindow(currentWindow, cam.lastMouseX, cam.lastMouseY);
202 SDL_SetWindowRelativeMouseMode(currentWindow, false);
203 }
204 wasRightMouseDown = rightMouseDown;
205
206 float radYaw = glm::radians(cam.yaw);
207 float radPitch = glm::radians(cam.pitch);
208
209 cam.forward.x = cos(radYaw) * cos(radPitch);
210 cam.forward.y = sin(radPitch);
211 cam.forward.z = sin(radYaw) * cos(radPitch);
212 cam.forward = amath::normalize(cam.forward);
213
214 cam.right = amath::normalize(amath::cross(cam.forward, glm::vec3(0.0f, 1.0f, 0.0f)));
215
216 const bool* keyboard = SDL_GetKeyboardState(nullptr);
217 glm::vec3 moveDir(0.0f);
218
219 if (keyboard[SDL_SCANCODE_W]) moveDir += cam.forward;
220 if (keyboard[SDL_SCANCODE_S]) moveDir -= cam.forward;
221 if (keyboard[SDL_SCANCODE_A]) moveDir -= cam.right;
222 if (keyboard[SDL_SCANCODE_D]) moveDir += cam.right;
223
224 if (glm::length(moveDir) > 0.0f) {
225 moveDir = glm::normalize(moveDir);
226 cam.cameraPos += moveDir * cam.speed * SystemManager::s_deltaTime;
227 }
228
229 cam.target = cam.cameraPos + cam.forward;
230 cam.viewMatrix = amath::lookAt(cam.cameraPos, cam.target, vec3(0.0f, 1.0f, 0.0f));
231 }
