62 bool containsIgnoreCase(const std::string& haystack, const std::string& needle) {
63 if (needle.empty()) return true;
64 auto it = std::search(
65 haystack.begin(), haystack.end(),
66 needle.begin(), needle.end(),
67 [](char ch1, char ch2) { return std::tolower(static_cast<unsigned char>(ch1)) == std::tolower(static_cast<unsigned char>(ch2)); }
68 );
69 return it != haystack.end();
